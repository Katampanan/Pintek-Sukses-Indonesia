                        const progress = Math.min(elapsed / duration, 1);
                        const eased = 1 - Math.pow(1 - progress, 3);
                        current = Math.floor(eased * target);
                        entry.target.textContent = current;
                        
                        if (progress < 1) {
                            requestAnimationFrame(animate);
                        } else {
                            entry.target.textContent = target;
                        }
                    };
                    
                    requestAnimationFrame(animate);
                    counterObserver.unobserve(entry.target);
                }
            });
        }, { threshold: 0.5 });
        
        counters.forEach(c => counterObserver.observe(c));
        
        // ============================================
        // ACTIVE NAV LINK ON SCROLL
        // ============================================
        const sections = document.querySelectorAll('section[id]');
        const navLinks = document.querySelectorAll('.nav-link');
        
        const sectionObserver = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    const id = entry.target.getAttribute('id');
                    navLinks.forEach(link => {
                        link.classList.remove('active');
                        if (link.getAttribute('href') === '#' + id) {
                            link.classList.add('active');
                        }
                    });
                }
            });
        }, { threshold: 0.3, rootMargin: '-80px 0px -50% 0px' });
        
        sections.forEach(s => sectionObserver.observe(s));
        
        // ============================================
        // CONTACT FORM VALIDATION
        // ============================================
        const form = document.getElementById('contact-form');
        const formSuccess = document.getElementById('form-success');
        const formError = document.getElementById('form-error');
        const formErrorDetail = document.getElementById('form-error-detail');
        
        form.addEventListener('submit', (e) => {
            e.preventDefault();
            
            formSuccess.classList.add('hidden');
            formError.classList.add('hidden');
            
            const fields = {
                name: { el: document.getElementById('name'), required: true, type: 'text' },
                company: { el: document.getElementById('company'), required: false, type: 'text' },
                email: { el: document.getElementById('email'), required: true, type: 'email' },
                phone: { el: document.getElementById('phone'), required: false, type: 'tel' },
                service: { el: document.getElementById('service'), required: true, type: 'select' },
                message: { el: document.getElementById('message'), required: true, type: 'text' }
            };
            
            let isValid = true;
            let firstError = '';
            
            Object.keys(fields).forEach(key => {
                const field = fields[key];
                const value = field.el.value.trim();
                
                field.el.classList.remove('error');
                
                if (field.required && !value) {
                    field.el.classList.add('error');
                    isValid = false;
                    if (!firstError) firstError = `${key.charAt(0).toUpperCase() + key.slice(1)} is required.`;
                } else if (field.type === 'email' && value) {
                    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
                    if (!emailRegex.test(value)) {
                        field.el.classList.add('error');
                        isValid = false;
                        if (!firstError) firstError = 'Please enter a valid email address.';
                    }
                }
            });
            
            if (!isValid) {
                formErrorDetail.textContent = firstError || 'Some required fields are missing or invalid.';
                formError.classList.remove('hidden');
                formError.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
                return;
            }
            
            if (CONFIG.formEndpoint) {
                fetch(CONFIG.formEndpoint, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        name: fields.name.el.value,
                        company: fields.company.el.value,
                        email: fields.email.el.value,
                        phone: fields.phone.el.value,
                        service: fields.service.el.value,
                        message: fields.message.el.value
                    })
                })
                .then(response => {
                    if (response.ok) {
                        showFormSuccess();
                    } else {
                        throw new Error('Submission failed');
                    }
                })
                .catch(error => {
                    formErrorDetail.textContent = 'There was an error submitting your inquiry. Please try again or contact us directly.';
                    formError.classList.remove('hidden');
                });
            } else {
                showFormSuccess();
            }
        });
        
        function showFormSuccess() {
            formSuccess.classList.remove('hidden');
            form.reset();
            formSuccess.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
            
            setTimeout(() => {
                formSuccess.classList.add('hidden');
            }, 8000);
        }
